<script>

// Add your YouTube Data API key here.
// Keep production API keys on a secure backend.
const API_KEY = "YOUR_YOUTUBE_API_KEY";

const moodSearches = {
    Relaxing: "relaxing peaceful music",
    Happy: "happy upbeat music",
    Studying: "study instrumental focus music",
    Workout: "energetic workout music",
    Emotional: "emotional soft music",
    Night: "chill nighttime music",
    Gospel: "Christian gospel worship music",
    Romantic: "romantic love songs"
};

// Select a mood
function selectMood(mood) {
    document.getElementById("searchInput").value =
        moodSearches[mood];

    searchMusic();
}

// Search YouTube
async function searchMusic() {

    const query = document
        .getElementById("searchInput")
        .value.trim();

    const status = document.getElementById("status");
    const results = document.getElementById("results");

    if (!query) {
        status.textContent = "Please enter a song or artist.";
        return;
    }

    if (API_KEY === "YOUR_YOUTUBE_API_KEY") {
        status.textContent =
            "Add your YouTube API key to activate music search.";
        return;
    }

    status.textContent = "Searching for music...";
    results.innerHTML = "";

    try {

        const url =
        "https://www.googleapis.com/youtube/v3/search" +
        "?part=snippet&type=video&videoEmbeddable=true" +
        "&maxResults=12&q=" + encodeURIComponent(query) +
        "&key=" + API_KEY;

        const response = await fetch(url);

        if (!response.ok) {
            throw new Error("YouTube search failed.");
        }

        const data = await response.json();

        if (!data.items || data.items.length === 0) {
            status.textContent = "No songs found.";
            return;
        }

        status.textContent =
            "Music results for: " + query;

        data.items.forEach(item => {

            const videoId = item.id.videoId;
            const title = item.snippet.title;
            const thumbnail =
                item.snippet.thumbnails.medium.url;

            const card = document.createElement("div");
            card.className = "song";

            const image = document.createElement("img");
            image.src = thumbnail;
            image.alt = "Music thumbnail";

            const heading = document.createElement("h3");
            heading.textContent = title;

            const button = document.createElement("button");
            button.textContent = "▶ Play Music";

            button.onclick = () => playMusic(videoId);

            card.appendChild(image);
            card.appendChild(heading);
            card.appendChild(button);

            results.appendChild(card);
        });

    } catch (error) {
        status.textContent =
            "Unable to search music. Please try again.";
        console.error(error);
    }
}

// Play selected YouTube music
function playMusic(videoId) {

    document.getElementById("youtubePlayer").src =
        "https://www.youtube-nocookie.com/embed/" +
        encodeURIComponent(videoId) +
        "?autoplay=1&rel=0";

    document.getElementById("status").textContent =
        "Now playing your selected music.";

    document.getElementById("music").scrollIntoView({
        behavior: "smooth"
    });
}
